### 2.1 假 SHAP 的生成方式

`scripts/csv_to_sql.py:233`:
```python
z = (val - mu) / sd
shap = z * 0.04 * direction   # 纯统计模拟，非模型输出
shap = max(-0.15, min(0.15, shap))
```

```python
import re

def _auto_name_mapping(raw_col: str) -> tuple[str, str]:
    zh_chars = re.findall(r'[\u4e00-\u9fff]+', raw_col)
    chinese_name = ''.join(zh_chars) if zh_chars else raw_col
    en_part = re.sub(r'[\u4e00-\u9fff]+', '', raw_col)
    en_part = re.sub(r'[^a-zA-Z0-9]', '_', en_part)
    en_part = re.sub(r'_+', '_', en_part).strip('_').lower()
    return (en_part or raw_col.lower(), chinese_name)
```

**b) 添加 `get_all_shap_factors()` 方法**

```python
def get_all_shap_factors(self) -> dict:
    if self._shap_cache is not None:
        return self._shap_cache

    row_df = self.df_raw.iloc[[-1]]
    last_date = str(self.df_raw.index[-1].date())

    # 构建特征矩阵
    X = pd.DataFrame(index=[0])
    for col in self.xgb_features:
        X[col] = float(row_df[col].iloc[0]) if col in row_df.columns else 0.0

    # 全量 SHAP
    sv = self.xgb_explainer.shap_values(X)
    shap_vals = sv[0] if isinstance(sv, list) else sv[0]

    # 真实风险分（复用 Stacking）
    loc = len(self.df_feat) - 1
    current_price = float(self.df_feat["price"].iloc[-1])
    lstm_pred, _ = self._lstm_predict_single(self.df_feat, loc)
    lstm_ret = (lstm_pred - current_price) / current_price
    xgb_risk = self._xgb_predict(row_df)
    market_state = self.df_raw[MARKET_STATE_COLS].iloc[-1].fillna(0).values.astype(float)
    pred_pct, _ = self._stacking_predict(market_state, lstm_ret, xgb_risk)
    risk_score = max(0, min(100, 50 - pred_pct * 10))

    factors = []
    for i, col in enumerate(self.xgb_features):
        en_key, zh_name = FEATURE_NAME_ZH.get(col, _auto_name_mapping(col))
        factors.append({
            "name": en_key,
            "nameZh": zh_name,
            "shapValue": round(float(shap_vals[i]), 6),
            "value": round(float(X[col].iloc[0]), 4),
            "category": _classify_factor(col),
        })
    factors.sort(key=lambda f: abs(f["shapValue"]), reverse=True)

    result = {"date": last_date, "riskScore": round(risk_score, 2), "factors": factors}
    self._shap_cache = result
    return result
```

### 3.3 Java 后端改动

#### 文件：`src/.../service/impl/FactorServiceImpl.java`

**完整重写逻辑：**

```java
@Slf4j
@Service
@RequiredArgsConstructor
public class FactorServiceImpl implements FactorService {

    private final FactorMapper factorMapper;
    private final RiskMapper riskMapper;
    private final WeightsConfig weightsConfig;
    private final RestTemplate restTemplate;

    @Value("${python.engine.url}")
    private String pythonEngineUrl;

    private volatile CachedShapResult cachedShap;

    // ── 内部缓存类 ──
    private static class CachedShapResult {
        final String date;
        final double riskScore;
        final List<RiskFactor> factors;
        final long timestamp;

        CachedShapResult(String date, double riskScore, List<RiskFactor> factors) {
            this.date = date;
            this.riskScore = riskScore;
            this.factors = factors;
            this.timestamp = System.currentTimeMillis();
        }

        boolean isExpired() {
            return System.currentTimeMillis() - timestamp > 300_000; // 5 min
        }
    }

    // ── 调用 Python Engine ──
    @SuppressWarnings("unchecked")
    private CachedShapResult fetchRealShapFactors() {
        CachedShapResult cached = this.cachedShap;
        if (cached != null && !cached.isExpired()) return cached;

        try {
            ResponseEntity<Map<String, Object>> resp = restTemplate.exchange(
                pythonEngineUrl + "/shap/factors",
                HttpMethod.GET, null,
                new ParameterizedTypeReference<>() {}
            );
            Map<String, Object> body = resp.getBody();
            if (body == null) return null;

            String date = (String) body.get("date");
            double riskScore = ((Number) body.get("riskScore")).doubleValue();
            List<Map<String, Object>> rawList =
                (List<Map<String, Object>>) body.get("factors");

            List<RiskFactor> factors = new ArrayList<>();
            for (Map<String, Object> f : rawList) {
                RiskFactor rf = new RiskFactor();
                rf.setDate(LocalDate.parse(date));
                rf.setFactorName((String) f.get("name"));
                rf.setFactorNameZh((String) f.get("nameZh"));
                rf.setCategory((String) f.get("category"));
                rf.setValue(BigDecimal.valueOf(((Number) f.get("value")).doubleValue()));
                rf.setShapValue(BigDecimal.valueOf(((Number) f.get("shapValue")).doubleValue()));
                factors.add(rf);
            }

            CachedShapResult result = new CachedShapResult(date, riskScore, factors);
            this.cachedShap = result;
            return result;
        } catch (RestClientException e) {
            log.warn("Python Engine /shap/factors 不可用, 降级到数据库: {}", e.getMessage());
            return null;
        }
    }

    // ── getRadarScores: 优先 Python，fallback DB ──
    @Override
    public List<RadarScoreVO> getRadarScores(LocalDate date) {
        List<RiskFactor> factors = null;

        if (date == null) {
            CachedShapResult py = fetchRealShapFactors();
            if (py != null) {
                factors = py.factors;
                date = LocalDate.parse(py.date);
            }
        } else {
            CachedShapResult py = fetchRealShapFactors();
            if (py != null && date.toString().equals(py.date)) {
                factors = py.factors;
            }
        }

        if (factors == null) {
            if (date == null) date = factorMapper.findLatestDate();
            if (date == null) throw new BusinessException(404, "No factor data");
            factors = factorMapper.findByDate(date);
        }
        if (factors.isEmpty()) throw new BusinessException(404, "No factor data for: " + date);

        // 分组 → 加权 → 归一化（逻辑不变）
        return buildRadarScores(factors);
    }

    // ── getExplain: 同策略 ──
    @Override
    public List<FactorVO> getExplain(LocalDate date) {
        CachedShapResult py = fetchRealShapFactors();
        List<RiskFactor> factors;
        if (py != null && date.toString().equals(py.date)) {
            factors = py.factors;
        } else {
            factors = factorMapper.findByDate(date);
        }
        return factors.stream().map(this::toFactorVO).collect(Collectors.toList());
    }

    // ── updateWeights: 基础分数来自真实模型 ──
    @Override
    public WeightUpdateResultVO updateWeights(WeightDTO dto) {
        validateWeights(dto);
        weightsConfig.setSupplyDemand(dto.getSupplyDemand());
        weightsConfig.setMacro(dto.getMacro());
        weightsConfig.setFinancial(dto.getFinancial());
        weightsConfig.setGeopolitical(dto.getGeopolitical());
        weightsConfig.setSentiment(dto.getSentiment());

        List<RiskFactor> factors;
        double baseRiskIndex;
        LocalDate latestDate;

        CachedShapResult py = fetchRealShapFactors();
        if (py != null) {
            factors = py.factors;
            baseRiskIndex = py.riskScore;
            latestDate = LocalDate.parse(py.date);
        } else {
            RiskIndex latest = riskMapper.findLatest();
            if (latest == null) throw new BusinessException(404, "No risk data");
            latestDate = latest.getDate();
            factors = factorMapper.findByDate(latestDate);
            baseRiskIndex = latest.getRiskIndex().doubleValue();
        }

        // 加权比率调整
        double weighted = 0, unweighted = 0;
        for (RiskFactor f : factors) {
            double abs = Math.abs(f.getShapValue().doubleValue());
            weighted += abs * weightsConfig.getWeight(f.getCategory());
            unweighted += abs;
        }
        double ratio = unweighted == 0 ? 1.0 : weighted / unweighted;
        double adjusted = Math.max(0, Math.min(100, baseRiskIndex * ratio));

        WeightUpdateResultVO result = new WeightUpdateResultVO();
        result.setRiskIndex(BigDecimal.valueOf(adjusted).setScale(2, RoundingMode.HALF_UP));
        result.setRiskLevel(RiskLevel.fromIndex(result.getRiskIndex()).getLabel());
        result.setRadarScores(getRadarScores(latestDate));
        result.setTopFactors(factors.stream()
            .sorted(Comparator.comparingDouble(
                (RiskFactor f) -> Math.abs(f.getShapValue().doubleValue())).reversed())
            .limit(5).map(this::toFactorVO).collect(Collectors.toList()));
        return result;
    }
}
```

## 4. SRS 需求缺口（对照需求规格说明书）

### 缺口 1：雷达图 Tooltip 缺少 Top-3 因子明细

**SRS UC-02 第 5 步**：「Hover 雷达轴线 → Tooltip 展示该类别下 Top-3 因子明细」

**现状**：`RadarChart.vue:32-39` 只配了默认 tooltip，API 返回的 `topFactors` 数据未使用。

**修复方向**：ECharts radar tooltip formatter 中读取 `topFactors`，展示因子名 + SHAP 值。

### 缺口 2：权重调整后没有展示风险指数

**SRS UC-03 第 5 步**：「更新后展示风险指数和等级」

**现状**：`FactorAnalysis.vue:80` 的 `await updateWeights(...)` 丢弃了返回值（包含 `riskIndex: 94.55, riskLevel: "High"`），页面上也没有展示风险指数的区域。

**修复方向**：
- 接收 PUT 响应中的 `riskIndex` 和 `riskLevel`
- 在页面顶部或权重滑块旁展示风险指数卡片

### 缺口 3：图表无过渡动画

**SRS UC-03 第 6 步**：「ECharts transition 动画（800ms cubicOut）」

**现状**：两个图表都用 `notMerge: true`（RadarChart.vue:90、FactorShapChart.vue:92），直接替换配置，无过渡效果。

**修复方向**：移除 `notMerge: true`，配置 `animationDuration: 800, animationEasing: 'cubicOut'`。

### 缺口 4：updateWeights 返回数据被浪费

**现状**：`FactorAnalysis.vue:80-88` 中，PUT 返回值包含完整的 `radarScores + topFactors`，但代码忽略了响应，又额外发了 2 次 GET 请求（radar + explain）。

**修复方向**：直接使用 PUT 响应中的 `radarScores` 和 `topFactors` 更新页面状态，省去 2 次网络请求。

## 5. 代码质量问题

### 5.1 雷达图 category 匹配脆弱

**位置**：`RadarChart.vue:26`

```javascript
d.category === 'Supply-Demand'  // 但 API 返回 "SUPPLY_DEMAND"
```

只靠第二条件 `d.categoryZh === t(...)` 的 i18n 兜底。翻译文本和后端 `categoryZh` 稍有出入就断。

**建议**：统一使用 `category` 大写枚举匹配（`SUPPLY_DEMAND`），或后端在返回时同时提供两种格式。

### 5.2 ECharts 全量引入

**位置**：RadarChart.vue、FactorShapChart.vue

```javascript
import * as echarts from 'echarts'  // ~1MB+
```

只需 radar、bar、tooltip 几个模块。

**建议**：改为按需引入：
```javascript
import { use } from 'echarts/core'
import { RadarChart, BarChart } from 'echarts/charts'
import { TooltipComponent, GridComponent } from 'echarts/components'
import { CanvasRenderer } from 'echarts/renderers'
```

### 5.3 初始请求串行

**位置**：`FactorAnalysis.vue` onMounted

`getCurrentRisk()` 先等完，再并行请求 radar + explain。但 `getRadarScores()` 不依赖日期参数（后端默认取最新），可以一起并行。
